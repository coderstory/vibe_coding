    在 `app-vue/src/composables/useHardwareMetrics.ts` 中创建 SSE 连接管理 composable。

    严格按照 D-07 和 D-08 实现，支持 mock EventSource 工厂注入（TDD-05）。

    ```typescript
    import { onUnmounted, ref } from 'vue'
    import type { HardwareMetricsDTO } from '@/api/modules/hardware'

    export interface UseHardwareMetricsOptions {
      /** EventSource 工厂函数，用于测试时注入 mock（TDD-05） */
      createEventSource?: (url: string) => EventSource
      /** SSE 服务端点 URL */
      url?: string
    }

    export function useHardwareMetrics(options: UseHardwareMetricsOptions = {}) {
      const {
        createEventSource = (url: string) => new EventSource(url),
        url = '/api/monitor/hardware/subscribe'
      } = options

      const connected = ref(false)
      const metrics = ref<HardwareMetricsDTO | null>(null)
      const error = ref<string | null>(null)
      let eventSource: EventSource | null = null
      let reconnectTimer: ReturnType<typeof setTimeout> | null = null
      let retryDelay = 1000
      const MAX_RETRY_DELAY = 30000

      function connect() {
        disconnect()

        eventSource = createEventSource(url)

        eventSource.addEventListener('connected', () => {
          connected.value = true
          error.value = null
          retryDelay = 1000  // reset on successful connection
        })

        eventSource.addEventListener('metrics', (event: MessageEvent) => {
          try {
            metrics.value = JSON.parse(event.data)
            error.value = null
          } catch (e) {
            error.value = '解析监控数据失败'
          }
        })

        eventSource.onerror = () => {
          connected.value = false
          eventSource?.close()
          scheduleReconnect()
        }
      }

      function scheduleReconnect() {
        error.value = `连接断开，${retryDelay / 1000} 秒后重连...`
        reconnectTimer = setTimeout(() => {
          retryDelay = Math.min(retryDelay * 2, MAX_RETRY_DELAY)
          connect()
        }, retryDelay)
      }

      function disconnect() {
        if (reconnectTimer) {
          clearTimeout(reconnectTimer)
          reconnectTimer = null
        }
        eventSource?.close()
        eventSource = null
        connected.value = false
      }

      function reconnect() {
        retryDelay = 1000
        connect()
      }

      onUnmounted(disconnect)

      return { connect, disconnect, reconnect, connected, metrics, error }
    }
    ```

    关键实现约束：
    - **指数退避严格遵循 D-08：** 1s -> 2s -> 4s -> 8s -> 16s -> max 30s。`scheduleReconnect()` 每次先乘以 2 然后 `Math.min` 截断到 MAX_RETRY_DELAY
    - **重连重置：** 收到 `connected` 事件时 `retryDelay` 重置为 1000
    - **手动重连：** `reconnect()` 将 `retryDelay` 重置为 1000 然后调用 `connect()`
    - **组件卸载：** `onUnmounted(disconnect)` 清理 EventSource 和定时器
    - **不暴露 EventSource 实例**（仅通过 `connected`/`error` 反映状态，符合 D-04 数据直接更新原则）
    - **createEventSource 可选参数：** 测试时传入工厂函数返回 mock EventSource（TDD-05）
    - **注释规范：** L2 级（鼓励）— 每条返回属性 JSDoc，关键逻辑（指数退避、重置规则）行内注释

    **注意：** `EventSource` 类型中 `addEventListener` 的 `event` 参数类型在 TS DOM lib 中可能需要显式类型断言。使用 `(event: MessageEvent)` 签名，如果 TS 报类型不兼容，使用 `(event: Event)` 并在内部转换为 `MessageEvent`。

    使用 mock EventSource（TDD-05）验证 composable 的核心行为。测试模式参考 RESEARCH.md §Mock EventSource for Vitest Testing。

    ```typescript
    import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest'
    import { useHardwareMetrics } from '@/composables/useHardwareMetrics'

    /** mock EventSource 实现 */
    class MockEventSource {
      private handlers: Map<string, (event: any) => void> = new Map()
      onerror: ((event: any) => void) | null = null
      onopen: ((event: any) => void) | null = null

      constructor(private _url: string) {
        // 模拟异步连接建立
        setTimeout(() => {
          const handler = this.handlers.get('connected')
          if (handler) handler({})
        }, 0)
      }

      addEventListener(event: string, handler: (event: any) => void) {
        this.handlers.set(event, handler)
      }

      close() {}

      // 测试辅助方法：模拟接收 SSE 事件
      mockReceiveEvent(event: string, data: string) {
        const handler = this.handlers.get(event)
        if (handler) handler({ data })
      }

      // 测试辅助方法：模拟连接错误
      mockError() {
        if (this.onerror) this.onerror({})
      }
    }

    describe('useHardwareMetrics', () => {
      let mockEs: MockEventSource

      beforeEach(() => {
        vi.useFakeTimers()
        mockEs = new MockEventSource('')
      })

      afterEach(() => {
        vi.useRealTimers()
      })

      it('应在调用 connect() 后建立连接并更新 connected 为 true', async () => {
        const { connect, connected } = useHardwareMetrics({
          createEventSource: () => mockEs as unknown as EventSource
        })

        connect()
        // 等待 connected 事件被触发
        await vi.runAllTimersAsync()
        expect(connected.value).toBe(true)
      })

      it('应在收到 metrics 事件后解析并更新 data', () => {
        const { connect, metrics } = useHardwareMetrics({
          createEventSource: () => mockEs as unknown as EventSource
        })

        connect()
        mockEs.mockReceiveEvent('metrics', JSON.stringify({
          timestamp: 1000,
          cpu: { systemLoad: 50.0, perCoreLoad: [50], logicalCores: 4, physicalCores: 4, processorName: 'Test', processorFrequency: 3000000000, cpu64bit: true }
        }))

        expect(metrics.value).not.toBeNull()
        expect(metrics.value?.cpu?.systemLoad).toBe(50.0)
        expect(metrics.value?.timestamp).toBe(1000)
      })

      it('应在连接错误时触发重连（指数退避首次为 1s）', async () => {
        const { connect, connected, error } = useHardwareMetrics({
          createEventSource: () => mockEs as unknown as EventSource
        })

        connect()
        await vi.runAllTimersAsync()
        // 首次连接成功
        expect(connected.value).toBe(true)

        // 模拟断线
        mockEs.mockError()
        // onerror 中设置 connected=false 并触发重连调度
        expect(connected.value).toBe(false)
        expect(error.value).toContain('重连')
      })

      it('应通过 reconnect() 重置重连延迟并连接', async () => {
        const { connect, reconnect, connected } = useHardwareMetrics({
          createEventSource: () => mockEs as unknown as EventSource
        })

        // 先断开一次让 retryDelay 增长
        connect()
        await vi.runAllTimersAsync()
        mockEs.mockError()

        // 手动 reconnect 应重置 delay 到 1s
        mockEs = new MockEventSource('')
        reconnect()
        await vi.runAllTimersAsync()
        expect(connected.value).toBe(true)
      })

      it('应通过 disconnect() 关闭连接并清理定时器', () => {
        const { connect, disconnect, connected } = useHardwareMetrics({
          createEventSource: () => mockEs as unknown as EventSource
        })

        connect()
        disconnect()
        expect(connected.value).toBe(false)
      })
    })
    ```

    关键实现约束：
    - **使用 `vi.useFakeTimers()` + `vi.runAllTimersAsync()`** 处理 setTimeout 异步调度，确保测试不挂起
    - **每个 `it` 块测试一个独立行为**（连接、数据接收、重连调度、手动重连、断开）
    - **`afterEach` 中 `vi.useRealTimers()`** 恢复真实定时器，避免跨测试污染
    - **mock EventSource 不依赖真实网络**，全在内存中模拟
    - **注释规范：** L1 级（建议）— 每个测试用例简要描述测试行为
